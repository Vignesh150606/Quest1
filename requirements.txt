# --- OCR stack ---
# Pinned per CLAUDE.md's "Known Gotchas": videocr-PaddleOCR's wrapper breaks on
# PaddleOCR 3.x. Verified: 2.10.0 resolves and imports cleanly.
paddleocr==2.10.0

# paddleocr does NOT declare paddlepaddle as a dependency, but imports it at module
# load (`from paddle.utils import try_import`), so `import paddleocr` fails with
# ModuleNotFoundError: No module named 'paddle' unless it's installed explicitly.
# 3.0.0 is the oldest 3.x that paddleocr 2.10.0 officially supports and works with the
# numpy 2.x that the rest of this stack pulls in (paddlepaddle 2.6.2 is built against
# numpy 1.x and would force a downgrade).
paddlepaddle==3.0.0

# videocr-PaddleOCR (the wrapper Phase 0 originally pinned around PaddleOCR) is
# dropped as of Phase 3: it does its own frame sampling and emits SRT, which is a poor
# fit for "find the onset of one target phrase" and gave no real control over sampling
# or onset detection. src/ocr_track.py calls PaddleOCR directly instead, per CLAUDE.md's
# "prefer a library's direct API over a wrapper when the wrapper adds little value."
# Dropping it also removes the constraint that forced paddleocr==2.10.0 -- that pin is
# kept anyway since 2.10.0 is the version actually verified against this codebase.
